{%- comment -%} Titles may carry a " | ..." suffix (e.g. a Hacker News link). jekyll-seo-tag has no title override, so render it, then swap the formatted full title for the clean one. {%- endcomment -%} {%- assign clean_title = page.title | split: " | " | first | strip -%} {%- capture seo_html -%}{%- seo -%}{%- endcapture -%} {%- if page.title and clean_title != page.title -%} {%- assign seo_full = page.title | markdownify | strip_html | normalize_whitespace | escape_once -%} {%- assign seo_clean = clean_title | markdownify | strip_html | normalize_whitespace | escape_once -%} {{- seo_html | replace: seo_full, seo_clean -}} {%- else -%} {{- seo_html -}} {%- endif %} {%- feed_meta -%} {%- if page.mathjax or site.mathjax -%} {%- endif -%}
